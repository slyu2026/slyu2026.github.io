\section{Injective and projective dimension}

\begin{remark}
It is not difficult to prove the following by transfinite induction (Baer's criterion). Let $A$ be a ring and $E\in\D^\heartsuit(A)$. Then $E$ is injective if and only if
\[
\Ext_A^1(A/\fa,E)=0
\qquad\text{for every ideal }\fa\subseteq A.
\]
Another transfinite induction argument tells us that every $M\in\D^\heartsuit(A)$ is contained in some injective $E\in\D^\heartsuit(A)$.

For a complex $X\in\D^+(A)$ and $\alpha,\beta\in\mathbb Z$, it is not difficult to show that the following are equivalent:
\begin{enumerate}[label=(\arabic*)]
\item For every $M\in\D^\heartsuit(A)$, we have
\[
\RHom_A(M,X)\in\D^{[\alpha,\beta]}(A).
\]
\item $X$ is represented by a complex of injective $A$-modules in degrees $[\alpha,\beta]$.
\end{enumerate}
In this case, we say that $X$ has \emph{injective amplitude} in $[\alpha,\beta]$.
\end{remark}

\medskip
\noindent\textbf{Problem 1.}
Let $(A,\m,k)$ be a Noetherian local ring.
\begin{enumerate}[label=(\arabic*)]
\item Let $X\in\Dcomp^+(A)$, $\Dtors^+(A)$, or $\Dcoh^+(A)$, and let $\alpha,\beta\in\mathbb Z$. Show that if
\[
\RHom_A(k,X)\in\D^{[\alpha,\beta]}(A),
\]
then $X$ has injective amplitude in $[\alpha,\beta]$.

In the $\Dtors^+(A)$ case, you are allowed to take the following statement for granted: if $I\in\D^\heartsuit(A)$ is injective, then $\RGamma_{\m}I$ is also in $\D^\heartsuit(A)$ and is injective.

\item Let $X\in\Dcomp^+(A)$ or $\Dcoh^+(A)$ have injective amplitude in $[\alpha,\beta]$. Show that, in fact,
\[
\RHom_A(M,X)\in\D^{[\alpha,\beta-p]}(A)
\]
for every $M\in\Dcoh^\heartsuit(A)$ of depth $p$.
\end{enumerate}

For $M\in\D^\heartsuit(A)$, let
\begin{align*}
\operatorname{pd}_A M
&=\min\bigl\{j\mid \text{for all }i>j,\ \Ext_A^i(M,-)=0\bigr\},\\
% The scan writes "max" in the following definition; retained verbatim.
\id_A M
&=\max\bigl\{j\mid \text{for all }i>j,\ \Ext_A^i(-,M)=0\bigr\}.
\end{align*}
Let $(A,\m,k)$ be Noetherian local. In the October~27 meeting, we saw, using minimal resolutions, that if $M\in\Dcoh^\heartsuit(A)$, then
\[
% The scan appears to write "min" here; retained rather than corrected to "max".
\operatorname{pd}_A M
=\min\bigl\{j\mid\Tor_j^A(k,M)\ne0\bigr\}.
\]
From the previous problem we see that if $M\in\D_\square^\heartsuit(A)$, where $\square\in\{\mathrm{tors},\mathrm{comp},\mathrm{coh}\}$, then
\[
% "min" is as written in the scan.
\id_A M=\min\bigl\{j\mid\Ext_A^j(k,M)\ne0\bigr\}.
\]

\medskip
\noindent\textbf{Problem 2.}
Let $f\in\m$ be a nonzerodivisor in $A$. Let $M\in\Dcoh^\heartsuit(A)$ be such that $fM=0$. Show that
\begin{align*}
% Both equalities are transcribed as written, including the direction of the shifts.
\operatorname{pd}_{A/fA}M&=\operatorname{pd}_A M+1,\\
\id_{A/fA}M&=\id_A M+1.
\end{align*}

\medskip
\noindent\textbf{Problem 3.}
Let $0\ne M\in\Dcoh^\heartsuit(A)\cap\Dtors(A)$, i.e.\ let $M$ be a nonzero finite-length $A$-module.
\begin{enumerate}[label=(\arabic*)]
\item If $\operatorname{pd}_A M<\infty$, show that $A$ is Cohen--Macaulay and $\operatorname{pd}_A M=\dimn A$.
\item If $\id_A M<\infty$, show that $A$ is Cohen--Macaulay and $\id_A M=\dimn A$.
\item Conversely, assume that $A$ is Cohen--Macaulay. Show that such an $M$ exists. You may take different modules $M$ for projective dimension and injective dimension, respectively.
\end{enumerate}
\noindent\emph{Hint.} For the projective dimension statement, use Auslander--Buchsbaum. For the injective dimension statement, consider $\Hom_A(M,E)$.
